## Before you begin

You need:

- A Cozmo AI account with an organization ([sign up](https://dash.prod.czmx.in))
- An API key - create one in **Dashboard - Organization Settings - API Keys**
- Your `org_id` - visible in the Dashboard URL and Organization Settings
- One of: `curl`, Python 3.8+, Node.js 18+, or Go 1.21+

<Info>
  All requests use the base URL `https://v2-nova-api.prod.czmx.in/api` with your API key in the `X-API-Key` header. All paths are scoped to your organization: `/org/{org_id}/...`.
</Info>

## Step 3 - Start a session

A session runs your agent against a prospect. You provide a client-generated UUID v4 as the `session_id` - this makes the operation idempotent.

<CodeGroup>

```bash cURL
# Generate a UUID: uuidgen (macOS/Linux) or PowerShell [guid]::NewGuid()
SESSION_ID=$(uuidgen | tr '[:upper:]' '[:lower:]')

curl -X POST "https://v2-nova-api.prod.czmx.in/api/org/{org_id}/agent-sessions/${SESSION_ID}/start" \
  -H "X-API-Key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "agentId": "agt-01j9xyz-abc123",
    "instruction": "Call Jane Doe at +19175551234. She submitted a support ticket about a billing issue on her account (ticket #4521). Help her understand why she was charged twice in March.",
    "title": "Support follow-up - Jane Doe - Ticket #4521",
    "prospectId": "5abb08f3-c2d7-44c1-9fb0-8bb7bb4aa66e"
  }'
```

## Step 5 - Set up a webhook

Instead of polling, receive a push notification the moment a session or call ends. Configure your endpoint in **Dashboard - Organization Settings - Webhooks**, then implement a handler:

<CodeGroup>

```javascript Node.js / Express
import express from "express";
import crypto from "crypto";

const app = express();

// Use raw body parser to preserve bytes for signature verification
app.post(
  "/webhooks/cozmo",
  express.raw({ type: "application/json" }),
  (req, res) => {
    const signature = req.headers["x-cozmo-signature"] as string;
    const secret = process.env.COZMO_WEBHOOK_SECRET!;

    // Verify signature
    const expected = crypto
      .createHmac("sha256", secret)
      .update(req.body)
      .digest("hex");

    if (!crypto.timingSafeEqual(Buffer.from(expected), Buffer.from(signature))) {
      return res.status(401).json({ error: "Invalid signature" });
    }

    // Acknowledge immediately
    res.json({ ok: true });

    // Process asynchronously
    const { event, data } = JSON.parse(req.body.toString());
    handleEvent(event, data).catch(console.error);
  }
);

async function handleEvent(event: string, data: any) {
  switch (event) {
    case "call.completed":
      console.log(`Call ${data.call_id} - Result: ${data.call_result}`);
      // Update your CRM, store transcript, trigger follow-up workflows
      break;

    case "call.failed":
      console.warn(`Call ${data.call_id} failed: ${data.call_result}`);
      // Log failure, schedule retry if appropriate
      break;

    case "session.completed":
      console.log(`Session ${data.session_id} completed in ${data.duration_seconds}s`);
      break;

    case "session.failed":
      console.warn(`Session ${data.session_id} failed: ${data.status}`);
      break;
  }
}

app.listen(3000, () => console.log("Listening on :3000"));
```

```python Python / FastAPI
import hashlib
import hmac
import os
from fastapi import FastAPI, Request, HTTPException, BackgroundTasks

app = FastAPI()

async def handle_event(event: str, data: dict):
    if event == "call.completed":
        evals = {e["key"]: e for e in data.get("evaluations", [])}
        print(f"Call {data['call_id']} - {data['call_result']}")
        if evals.get("INTERESTED", {}).get("result_boolean"):
            print(f"Prospect {data['prospect']['ExternalId']} is interested!")
    elif event == "call.failed":
        print(f"Call {data['call_id']} failed: {data['call_result']}")
    elif event == "session.completed":
        print(f"Session {data['session_id']} completed in {data['duration_seconds']}s")
    elif event == "session.failed":
        print(f"Session {data['session_id']} failed: {data['status']}")

@app.post("/webhooks/cozmo")
async def cozmo_webhook(request: Request, background_tasks: BackgroundTasks):
    raw_body = await request.body()
    signature = request.headers.get("x-cozmo-signature", "")
    secret = os.environ["COZMO_WEBHOOK_SECRET"]

    expected = hmac.new(secret.encode(), raw_body, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(expected, signature):
        raise HTTPException(status_code=401, detail="Invalid signature")

    payload = await request.json()
    background_tasks.add_task(handle_event, payload["event"], payload["data"])

    return {"ok": True}
```

```go Go / net/http
package main

import (
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "fmt"
    "io"
    "log"
    "net/http"
    "os"
)

type Payload struct {
    Event string          `json:"event"`
    Data  json.RawMessage `json:"data"`
}

func verifySignature(body []byte, sig, secret string) bool {
    mac := hmac.New(sha256.New, []byte(secret))
    mac.Write(body)
    return hmac.Equal([]byte(hex.EncodeToString(mac.Sum(nil))), []byte(sig))
}

func handleEvent(event string, data json.RawMessage) {
    switch event {
    case "call.completed":
        var d struct {
            CallID     string `json:"call_id"`
            CallResult string `json:"call_result"`
        }
        json.Unmarshal(data, &d)
        log.Printf("Call %s - %s", d.CallID, d.CallResult)

    case "session.completed":
        var d struct {
            SessionID       string `json:"session_id"`
            DurationSeconds int    `json:"duration_seconds"`
        }
        json.Unmarshal(data, &d)
        log.Printf("Session %s completed in %ds", d.SessionID, d.DurationSeconds)
    }
}

func webhookHandler(w http.ResponseWriter, r *http.Request) {
    body, _ := io.ReadAll(r.Body)

    if !verifySignature(body, r.Header.Get("X-Cozmo-Signature"), os.Getenv("COZMO_WEBHOOK_SECRET")) {
        http.Error(w, "invalid signature", http.StatusUnauthorized)
        return
    }

    var p Payload
    json.Unmarshal(body, &p)

    // Respond first, process in background
    w.Header().Set("Content-Type", "application/json")
    w.Write([]byte(`{"ok":true}`))

    go handleEvent(p.Event, p.Data)
}

func main() {
    http.HandleFunc("/webhooks/cozmo", webhookHandler)
    fmt.Println("Webhook server on :3000")
    log.Fatal(http.ListenAndServe(":3000", nil))
}
```

</CodeGroup>

<Warning>
  Always verify the `X-Cozmo-Signature` header. Without this check, anyone can send fake events to your endpoint. Compute the HMAC-SHA256 over the **raw request body bytes** before any JSON parsing.
</Warning>

## Step 6 - Trigger a workflow run

For production outreach at scale, use workflows instead of direct sessions. A workflow run creates both the session and any associated calls according to your campaign configuration:

<CodeGroup>

```bash cURL
curl -X POST https://v2-nova-api.prod.czmx.in/api/org/{org_id}/workflow-runs \
  -H "X-API-Key: YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "workflow_id": "wf_abc123",
    "prospect": {
      "phone": "+19175551234",
      "first_name": "Jane",
      "last_name": "Doe",
      "email": "jane.doe@example.com",
      "external_id": "crm_789",
      "custom_data": {
        "deal_stage": "proposal",
        "contract_value": 45000
      }
    }
  }'
```

**Expected response:**

```json
{
  "workflow_run_id": "run_xyz789",
  "external_id": "crm_789"
}
```

<Tip>
  Set `external_id` to your CRM's contact ID. This value appears as `prospect.ExternalId` in webhook payloads, letting you map call outcomes back to the right record without any extra lookups.
</Tip>

Workflow runs also support scheduling. Add `"scheduled_at": "2026-04-22T09:00:00Z"` to the request body to delay execution until a specific time.
