## Official SDKs

Cozmo AI provides official SDKs generated from the Nova API OpenAPI specification using [Stainless](https://stainless.com). All SDKs are kept in sync with the API automatically and provide:

- Full type safety (TypeScript types, Python type hints, Go structs)
- Automatic retries with exponential backoff
- Consistent error handling
- Environment variable configuration

### Error handling

```python
from cozmoai import CozmoAI
from cozmoai.exceptions import (
    AuthenticationError,
    NotFoundError,
    RateLimitError,
    BadRequestError,
    APIError,
)

client = CozmoAI()

try:
    run = client.workflow_runs.create(
        org_id=ORG_ID,
        workflow_id="wf_abc123",
        prospect={"phone": "+19175551234"},
    )
except AuthenticationError:
    print("Invalid API key - check COZMO_API_KEY")
except BadRequestError as e:
    print(f"Validation error: {e.message}")  # e.g. "phone must be in E.164 format"
except NotFoundError:
    print("Workflow not found - check workflow_id")
except RateLimitError as e:
    print(f"Rate limited. Retry after: {e.retry_after}")
except APIError as e:
    print(f"API error {e.status_code}: {e.message}")
```

### Using in a FastAPI service

```python
# main.py
import os
import uuid
import logging
from contextlib import asynccontextmanager
from fastapi import FastAPI, HTTPException, BackgroundTasks, Request
from pydantic import BaseModel
from cozmoai import CozmoAI

log = logging.getLogger(__name__)
nova: CozmoAI = None


@asynccontextmanager
async def lifespan(app: FastAPI):
    global nova
    nova = CozmoAI()  # Reads COZMO_API_KEY from env
    log.info("Nova client initialized")
    yield


app = FastAPI(lifespan=lifespan)
ORG_ID = os.environ["ORG_ID"]


class TriggerCallRequest(BaseModel):
    workflow_id: str
    phone: str
    first_name: str
    external_id: str
    custom_data: dict = {}


@app.post("/trigger-call")
async def trigger_call(body: TriggerCallRequest):
    try:
        run = nova.workflow_runs.create(
            org_id=ORG_ID,
            workflow_id=body.workflow_id,
            prospect={
                "phone": body.phone,
                "first_name": body.first_name,
                "external_id": body.external_id,
                "custom_data": body.custom_data,
            },
        )
        return {"workflow_run_id": run.workflow_run_id}
    except Exception as e:
        log.error("trigger_call_failed", extra={"error": str(e)})
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/webhook/cozmo")
async def receive_webhook(request: Request, background: BackgroundTasks):
    payload = await request.json()
    background.add_task(process_webhook, payload)
    return {"ok": True}


async def process_webhook(payload: dict):
    event = payload.get("event")
    data = payload.get("data", {})

    if event == "call.completed":
        call_id = data.get("call_id")
        if call_id:
            call = nova.calls.get(org_id=ORG_ID, call_id=call_id)
            log.info("call_completed", extra={
                "call_id": call_id,
                "duration": call.duration_seconds,
                "cost": call.cost_usd,
            })
```

```typescript
import type {
  AgentResponse,
  AgentListResponse,
  CallDetailResponse,
  CallListResponse,
  SessionResponse,
  WorkflowResponse,
  WorkflowRunResponse,
  EvaluationResult,
  ToolLog,
} from "cozmoai";

async function getInterestingCalls(orgId: string): Promise<CallDetailResponse[]> {
  const client = new CozmoAI();
  const list: CallListResponse = await client.calls.list({ orgId, status: "FINISHED" });

  const detailed: CallDetailResponse[] = await Promise.all(
    list.data.map((c) => client.calls.get({ orgId, callId: c.id }))
  );

  return detailed.filter((call) =>
    call.evaluations?.some(
      (ev: EvaluationResult) => ev.outcome_key === "INTERESTED" && ev.result_boolean
    )
  );
}
```

### Error handling

```typescript
import CozmoAI, {
  AuthenticationError,
  NotFoundError,
  RateLimitError,
  BadRequestError,
  APIError,
} from "cozmoai";

const client = new CozmoAI();

async function safeCreateRun(workflowId: string, prospect: object) {
  try {
    return await client.workflowRuns.create({
      orgId: ORG_ID,
      workflowId,
      prospect,
    });
  } catch (err) {
    if (err instanceof AuthenticationError) {
      console.error("Invalid API key - check COZMO_API_KEY");
      throw err;
    }
    if (err instanceof BadRequestError) {
      console.error("Validation error:", err.message);
      throw err; // Don't retry 4xx
    }
    if (err instanceof NotFoundError) {
      console.error("Workflow not found:", workflowId);
      throw err;
    }
    if (err instanceof RateLimitError) {
      console.warn("Rate limited, retry after:", err.retryAfter);
      throw err;
    }
    if (err instanceof APIError) {
      console.error(`API error ${err.status}: ${err.message}`);
    }
    throw err;
  }
}
```

### Async patterns

```typescript
// Parallel workflow runs with controlled concurrency
import { default as pLimit } from "p-limit"; // npm install p-limit

const limit = pLimit(10); // Max 10 concurrent requests

async function triggerBulkRuns(prospects: object[]) {
  const results = await Promise.allSettled(
    prospects.map((prospect) =>
      limit(() =>
        client.workflowRuns.create({
          orgId: ORG_ID,
          workflowId: "wf_abc123",
          prospect,
        })
      )
    )
  );

  const succeeded = results.filter((r) => r.status === "fulfilled").length;
  const failed = results.filter((r) => r.status === "rejected").length;
  console.log(`Triggered: ${succeeded} succeeded, ${failed} failed`);
  return results;
}

// Async generator for paginating all calls
async function* getAllCalls(status?: string) {
  let page = 1;
  while (true) {
    const result = await client.calls.list({
      orgId: ORG_ID,
      status,
      page,
      size: 100,
    });
    for (const call of result.data) yield call;
    if (page >= result.meta.totalPages) break;
    page++;
  }
}

for await (const call of getAllCalls("FINISHED")) {
  console.log(call.id, call.duration_seconds);
}
```

### Using with Next.js App Router

```typescript
// app/api/trigger-call/route.ts
import { NextRequest, NextResponse } from "next/server";
import CozmoAI from "cozmoai";

// Initialize outside the handler for connection reuse
const nova = new CozmoAI({ apiKey: process.env.COZMO_API_KEY });
const ORG_ID = process.env.ORG_ID!;

export async function POST(req: NextRequest) {
  const { workflowId, phone, firstName, externalId } = await req.json();

  if (!workflowId || !phone) {
    return NextResponse.json({ error: "Missing required fields" }, { status: 400 });
  }

  try {
    const run = await nova.workflowRuns.create({
      orgId: ORG_ID,
      workflowId,
      prospect: { phone, firstName, externalId },
    });
    return NextResponse.json({ runId: run.workflowRunId });
  } catch (err: any) {
    console.error("trigger_call_error", err.message);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}
```

```go
package main

import (
    "context"
    "fmt"
    "log"
    "os"

    "github.com/cozmoai/cozmoai-go"
)

func main() {
    ctx := context.Background()
    orgID := os.Getenv("ORG_ID")
    client := cozmoai.NewClient()

    // Create agent
    agent, err := client.Agents.Create(ctx, orgID, &cozmoai.CreateAgentRequest{
        Name: "Outbound Sales Agent",
        Type: "voice",

    })
    if err != nil {
        log.Fatalf("create agent: %v", err)
    }
    fmt.Println("Created agent:", agent.ID)

    // Update configuration
    _, err = client.Agents.Update(ctx, orgID, agent.ID, &cozmoai.UpdateAgentRequest{
        LLMConfig: &cozmoai.LLMConfig{
            Provider:    "openai",
            Model:       "gpt-4o",
            Temperature: 0.7,
            MaxTokens:   512,
        },
        TranscriberConfig: &cozmoai.TranscriberConfig{
            Provider:      "deepgram",
            Model:         "nova-3",
            Language:      "en-US",
            EndpointingMs: 300,
            SmartFormat:   true,
        },
        VoiceConfig: &cozmoai.VoiceConfig{
            Provider: "elevenlabs",
            VoiceID:  "21m00Tcm4TlvDq8ikWAM",
            Model:    "eleven_turbo_v2",
        },
    })
    if err != nil {
        log.Fatalf("update agent: %v", err)
    }

    // List agents
    agents, err := client.Agents.List(ctx, orgID, &cozmoai.ListAgentsParams{
        Page: 1, Size: 20,
    })
    if err != nil {
        log.Fatal(err)
    }
    for _, a := range agents.Data {

    }

    // Get single agent
    detail, err := client.Agents.Get(ctx, orgID, agent.ID)

        log.Fatal(err)
    }
    fmt.Println("Agent:", detail.Name, "updated at", detail.UpdatedAt)
```

```go
import (

run, err := client.WorkflowRuns.Create(ctx, orgID, req)
if err != nil {
    var apiErr *cozmoai.APIError
    if errors.As(err, &apiErr) {
        switch apiErr.StatusCode {
        case 400:
            fmt.Println("Validation error:", apiErr.Message)
            // Don't retry - fix the request
        case 401:
            fmt.Println("Invalid API key")
        case 404:
            fmt.Println("Workflow not found")
        case 429:
            fmt.Println("Rate limited, retry after:", apiErr.RetryAfter)
        default:
            fmt.Printf("API error %d: %s\n", apiErr.StatusCode, apiErr.Message)
        }
        return
    }
    // Network or other error
    log.Fatal("request failed:", err)
}
```

### Goroutine-based bulk calling

```go
package main

import (
    "context"
    "fmt"
    "log"
    "sync"
    "time"

    "github.com/cozmoai/cozmoai-go"
)

func TriggerBulkRuns(orgID, workflowID string, prospects []cozmoai.ProspectData) ([]string, []error) {
    client := cozmoai.NewClient()
    ctx := context.Background()

    type result struct {
        runID string
        err   error
        idx   int
    }

    results := make([]result, len(prospects))
    sem := make(chan struct{}, 10) // Limit to 10 concurrent requests
    var wg sync.WaitGroup

    for i, prospect := range prospects {
        wg.Add(1)
        go func(idx int, p cozmoai.ProspectData) {
            defer wg.Done()
            sem <- struct{}{}
            defer func() { <-sem }()

            run, err := client.WorkflowRuns.Create(ctx, orgID, &cozmoai.CreateWorkflowRunRequest{
                WorkflowID: workflowID,
                Prospect:   p,
            })
            if err != nil {
                results[idx] = result{err: err, idx: idx}
                return
            }
            results[idx] = result{runID: run.WorkflowRunID, idx: idx}
        }(i, prospect)
    }

    wg.Wait()

    var runIDs []string
    var errs []error
    for _, r := range results {
        if r.err != nil {
            errs = append(errs, fmt.Errorf("prospect %d: %w", r.idx, r.err))
        } else {
            runIDs = append(runIDs, r.runID)
        }
    }

    fmt.Printf("Triggered %d runs, %d failed\n", len(runIDs), len(errs))
    return runIDs, errs
}
```

```go
// Production client with connection pool and custom HTTP client
import (
    "net/http"

    "github.com/cozmoai/cozmoai-go"

func NewProductionClient() *cozmoai.Client {
    httpClient := &http.Client{

        Transport: &http.Transport{
            MaxIdleConns:        100,
            MaxIdleConnsPerHost: 20,

    return cozmoai.NewClient(

        cozmoai.WithHTTPClient(httpClient),
        cozmoai.WithMaxRetries(3),
    )
}
```

## Using the REST API directly

If your language is not supported by an official SDK, use the REST API directly via HTTP. All examples throughout this documentation use `curl` and show raw HTTP requests.

- Base URL: `https://v2-nova-api.prod.czmx.in/api`
- Auth header: `X-API-Key: YOUR_API_KEY`
- All paths org-scoped: `/org/{org_id}/...`
- Content-Type: `application/json` for POST/PUT requests

See the [API Reference](/api-reference) for the complete endpoint list with interactive request builders.

## Environment variable reference

| Variable | Description |
|----------|-------------|
| `COZMO_API_KEY` | Your Nova API key (required) |
| `ORG_ID` | Your organization UUID (required for most operations) |

<Warning>
  Never commit API keys to source control. Use environment variables, a secrets manager (AWS Secrets Manager, HashiCorp Vault, Doppler), or a `.env` file excluded from git via `.gitignore`.
</Warning>
